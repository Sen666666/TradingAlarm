def save(p,x):
 p=Path(p);p.parent.mkdir(parents=True,exist_ok=True)

def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def fee(q,b):return sum(q.fees.incremental('buy',b,{}, {},ONE,shares=b/q.raw_price,par_value=q.par_value).values(),ZERO) if b>0 else ZERO

def allocate(items,cash):
 """Capped proportional water filling; all inputs frozen at previous close."""
 if not items or cash<=0:return {}
 caps={i['sid']:i['budget'] for i in items};weights={i['sid']:D(str(i['weight'])) for i in items};out={s:ZERO for s in caps};left=min(cash,sum(caps.values(),ZERO));active=set(caps)
 while active and left>0:
  weight=sum((weights[s] for s in active),ZERO);hit=[s for s in active if left*weights[s]/weight>=caps[s]-out[s]]
  if not hit:
   for s in active:out[s]+=left*weights[s]/weight
   break
  for s in sorted(hit):x=caps[s]-out[s];out[s]+=x;left-=x;active.remove(s)
 # Include estimated fees in the common scale. No ordering-based funding advantage.
 cost=lambda scale:sum((b*scale+fee(next(i['quote'] for i in items if i['sid']==s),b*scale) for s,b in out.items()),ZERO)
 low,high=(ONE,ONE) if cost(ONE)<=cash else (ZERO,ONE)
 for _ in range(70):
  mid=(low+high)/2
  if cost(mid)<=cash:low=mid
  else:high=mid
 return {s:(b*low).quantize(D('.01'),rounding='ROUND_FLOOR') for s,b in out.items()}

class Account(L.SectorAccount):
 def __init__(self,name,market,facts):
  super().__init__(fractional_split_policy=market.protocol['fractional_split_policy'])
  self.name=name;self.market=market;self.facts=facts;self.current_nav=INITIAL;self.layers=[];self.layer_seq=0;self.sell_layers={};self.order_meta={};self.watch={};self.core_epoch={};self.last_buy={};self.pending=[];self.decisions=[];self.trades=[];self.counts=collections.Counter();self.minimum_available=INITIAL
 def log(self,kind,at,**kw):self.audit_log.append(dict(kind=kind,at=at,**kw))
 def live(self,sid):return [x for x in self.layers if x['sid']==sid and x['qty']]
 def check(self):
  assert self.available>=0
  for sid in set(self.positions)|{x['sid'] for x in self.layers}:
   assert sum(x['qty'] for x in self.live(sid))==self.quantity(sid),(self.name,sid,'virtual/physical inventory')
 def fill_buy(self,sid,w):
  order=self.buys.get(sid)
  if order is None:return
  meta=self.order_meta[order.order_id];f=self.facts.factor(sid,w.at)
  if w.raw_open*f<=meta['floor']:
   self.counts['open_at_or_below_protection']+=1;self.log('pilot_buy_rejected',w.at,sid=sid,reason='open_at_or_below_protection',intent=meta);self.cancel_buy(sid);return
  receipt=super().fill_buy(sid,w)
  if receipt:
   self.layer_seq+=1
   z=dict(id=self.layer_seq,sid=sid,kind=meta['kind'],qty=receipt.shares,initial_qty=receipt.shares,opened=w.at,signal_at=meta['clock'].decision_at,epoch=meta['epoch'],floor=meta['floor'],cash=-receipt.gross_cny-receipt.fees_cny,dividend=ZERO,closed=None,reason=None)
   self.layers.append(z);self.last_buy[sid]=w.execution_price*f
   if z['kind']=='core':self.core_epoch[sid]=meta['epoch']
   self.trades.append(dict(**asdict(receipt),layer=z['id'],kind=z['kind'],signal_at=z['signal_at'],raw_open=w.raw_open,price=w.execution_price,floor=meta['floor'],pvr=meta['score'],reason=meta['reason'],epoch=meta['epoch']))
   self.counts['buy_fills']+=1
  self.check();return receipt
 def fill_exit(self,sid,w):
  receipt=super().fill_exit(sid,w)
  if receipt:
   left=receipt.shares;targets=self.sell_layers[sid];parts=[]
   for z in self.live(sid):
    if z['id'] not in targets:continue
    take=min(left,z['qty']);z['qty']-=take;left-=take;cash=(receipt.gross_cny-receipt.fees_cny)*D(take)/receipt.shares;z['cash']+=cash
    if not z['qty']:z['closed']=w.at
    parts.append(dict(layer=z['id'],kind=z['kind'],shares=take,net_cash=cash,reason=z['reason']))
    if not left:break
   assert not left,(sid,receipt,'sale allocation')
   self.trades.append(dict(**asdict(receipt),parts=parts,raw_open=w.raw_open,price=w.execution_price,reason=','.join(sorted({p['reason'] for p in parts}))))
   if sid not in self.exits:self.sell_layers.pop(sid,None)
   if not self.quantity(sid):self.last_buy.pop(sid,None)
   self.counts['sell_fills']+=1
  self.check();return receipt
 def split(self,**kw):
  sid=kw['sid'];layers=self.live(sid);old=sum(z['qty'] for z in layers);super().split(**kw)
  if old:
   target=self.quantity(sid);parts=[int(D(z['qty'])*kw['ratio']) for z in layers];remain=target-sum(parts)
   # Fractional entitlement writeoff belongs to the original aggregate ledger.
   for i in range(remain):parts[i%len(parts)]+=1
   for z,q in zip(layers,parts):z['qty']=q
   if sid in self.exits and self.exits[sid].remaining_target is not None:
    self.exits[sid].remaining_target=sum(z['qty'] for z in self.live(sid) if z['id'] in self.sell_layers[sid]);self.exits[sid].submitted_remaining=min(self.exits[sid].submitted_remaining,self.exits[sid].remaining_target)
  self.check()
 def accrue_dividend(self,**kw):
  super().accrue_dividend(**kw);layers=self.live(kw['sid']);qty=sum(z['qty'] for z in layers)
  assert qty
  for z in layers:z['dividend']+=kw['net_local']*D(z['qty'])/qty
 def sell(self,sid,clock,quote,ids,reason):
  ids=set(ids)
  for z in self.live(sid):
   if z['id'] in ids:z['reason']=reason
  ids|=self.sell_layers.get(sid,set());self.sell_layers[sid]=ids
  shares=sum(z['qty'] for z in self.live(sid) if z['id'] in ids)
  if not shares:return
  if sid not in self.exits:super().request_reduction(quote,clock,shares)
  else:
   ex=self.exits[sid];ex.remaining_target=shares
   if shares==self.quantity(sid):ex.purpose='structure_exit';ex.remaining_target=None
  self.log('pilot_layer_exit',clock.decision_at,sid=sid,layers=sorted(ids),shares=shares,reason=reason)

class Pilot(N.HistoryRunner):

 def planning(self,at):

     a.sell(sid,clock,quote,[x['id'] for x in layers],'ACT_pause' if not a.risk_state['on'] else 'I_structure_exit');continue

     if x['kind']!='add' or x['id'] in a.sell_layers.get(sid,set()):continue

   if a.risk_state['on']:
    for sid,s in sd.items():
     if sid in a.exits or sid in a.buys:continue
     r=self.facts.row(sid,at);c=cp[sid];p=score[sid];epoch=self.global_watch.get(sid);held=bool(a.quantity(sid));floor=self.facts.floor(sid,at,'I')
     if floor is None:continue

     if held and D(str(r.close))<=a.last_buy[sid]:confirmed=False

 def opening(self,a,at):
  for sid in sorted(a.exits):
   w=self.market.window(sid,at,'sell',ONE)
   if w:a.fill_exit(sid,w)
  a.settle_due(at)
  items=[i for i in a.pending if i['clock'].earliest_execution_at==at];cash=max(ZERO,a.available-a.current_nav*D('.1'));budgets={}
  # Both priority groups are allocated before any fill. Failed minimum/guard cash is not recycled in this opening.
  for group in (0,1):
   groupitems=[i for i in items if i['group']==group];z=allocate(groupitems,cash);budgets.update(z);cash-=sum((v+fee(next(i['quote'] for i in groupitems if i['sid']==s),v) for s,v in z.items()),ZERO)
  for i in sorted(items,key=lambda x:(x['group'],x['sid'])):
   sid=i['sid'];budget=budgets.get(sid,ZERO);a.log('pilot_frozen_funding',at,sid=sid,budget=budget,signal_at=i['clock'].decision_at,desired=i['budget'],layer_kind=i['kind'],group=i['group'])
   if budget<=0:a.counts['cash_unfunded']+=1;continue
   ok,oid=a.reserve_budget(quote=i['quote'],clock=i['clock'],gross_budget_cny=budget,purpose='annual_rebalance',target_shares_cap=i['cap'],candidate_id='BP1:'+i['reason'],reservation_at=at)
   if not ok:a.counts[oid]+=1;continue
   a.order_meta[oid]=i;w=self.market.window(sid,at,'buy',ONE)
   if w:a.fill_buy(sid,w)
   else:a.counts['no_open_window']+=1
  a.settle_due(at);a.minimum_available=min(a.minimum_available,a.available);a.check()
 def run(self,end='2026-10-27'):
  start=N.iso('2023-12-29T15:10:00+08:00');finish=N.iso(end+'T15:10:00+08:00');cutoffs={x for x in self.market.cutoffs if start<=x<=finish};opens={x for x in self.market.sessions['A'] if start<=x<=finish};acts={datetime.combine(datetime.fromisoformat(d).date(),time.min,N.TZ) for d in self.actions_by_date};events=sorted(cutoffs|opens|{x for x in acts if start<=x<=finish})
  assert finish in cutoffs
  for at in events:
   for a in self.accounts.values():a.settle_due(at)
   self._actions(at)
   for a in self.accounts.values():self._payments(a,at)
   if at in opens:
    for a in self.accounts.values():self.opening(a,at)
   if at in cutoffs:
    self.planning(at)
    if at.month in (1,7) and at.day<=3:print(str(at.date()),len(self.daily),flush=True)
  return self
