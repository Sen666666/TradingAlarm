def strict_calc(f,k):
 out=[]
 for _,g in f.groupby('segment_id',sort=False):

  for t,r in g.iterrows():
   exited=False
   if active and ev.compare(float(r.close),floor)<=0:active=False;exited=True
   if e.loc[t,'event']==1:
    pivot=float(e.loc[t,'confirmed_pivot_price'])
    if active:floor=max(floor,pivot)
    elif not exited:active=True;floor=pivot
   rows.append(dict(date=t,strict_target=active,strict_floor=floor if active or exited else math.nan,event=float(e.loc[t,'event'])))
  out.append(pd.DataFrame(rows).set_index('date'))
 return pd.concat(out) if out else pd.DataFrame({'strict_target':pd.Series(dtype=bool),'strict_floor':pd.Series(dtype=float),'event':pd.Series(dtype=float)},index=f.index[:0])
